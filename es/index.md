<!-- 04 · Especialidades técnicas -->
  <section id="expertise" class="arkhe-section" aria-labelledby="arkhe-expertise-title">

        </p>
      </div>
    </article>
    <article class="arkhe-tech-card">

        </p>
      </div>
    </article>
    <article class="arkhe-tech-card">

      <div class="arkhe-tech-content">
        <h3>Integración multi-ómica</h3>

        </p>
      </div>
    </article>
    <article class="arkhe-tech-card">
      <img src="{{ '/assets/images/omics/bulk-rnaseq.webp' | relative_url }}"

        <h3>Bulk RNA-seq</h3>

          interpretación funcional.
        </p>
